\section{Impl\'{e}mentation}

\begin{frame}{Mod\`{e}le de Donn\'{e}es (Extrait)}
    \begin{lstlisting}
class Course(models.Model):
    title = models.CharField(max_length=255)
    teacher = models.ForeignKey(Utilisateur, on_delete=models.CASCADE)
    enrolled_students = models.ManyToManyField(Utilisateur, blank=True)

class Progress(models.Model):
    student = models.ForeignKey(Utilisateur, on_delete=models.CASCADE)
    course = models.ForeignKey(Course, on_delete=models.CASCADE)
    completed_lessons = models.JSONField(default=list)
    quiz_scores = models.JSONField(default=dict)
    \end{lstlisting}
\end{frame}

\begin{frame}{S\'{e}curit\'{e} du Syst\`{e}me}
    \begin{itemize}
        \item \textbf{Mots de passe :} Hachage cryptographique PBKDF2.
        \item \textbf{Protection CSRF :} Jetons obligatoires sur tous les formulaires.
        \item \textbf{Protection XSS :} \'{E}chappement automatique des donn\'{e}es dans les templates.
        \item \textbf{Injection SQL :} Utilisation syst\'{e}matique de l'ORM Django (requ\^{e}tes param\'{e}tr\'{e}es).
        \item \textbf{Autorisation :} Contr\^{o}le d'acc\`{e}s granulaire via r\^{o}les personnalis\'{e}s.
    \end{itemize}
\end{frame}
